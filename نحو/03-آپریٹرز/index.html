    03 — آپریٹرز

    آپریٹرز اقدار پر مختلف عملیات انجام دیتے ہیں۔

    ریاضی کے آپریٹرز

    عمل الفاظ علامت
    جمع جمع، جوڑ +
    تفریق منفی، گھٹاؤ -
    ضرب ضرب *
    تقسیم ونڈ، تقسیم /
    باقی باقی %

    مثال

    پندھ_شروع "حساب"
    شے a = 10
    شے b = 5

    لکھو(a جمع b)
    لکھو(a منفی b)
    لکھو(a ضرب b)
    لکھو(a ونڈ b)
پندھ_مک_گیا

    متوقع نتائج:

    15
5
50
2

    موازنہ

    عمل الفاظ
    برابر برابر
    بڑا وڈا
    چھوٹا نکا

    مثال:

    اگر عمر وڈا 18 {
    لکھو("بالغ")
}

    منطقی آپریٹرز

    عمل لفظ
    AND تے
    OR یا
    NOT کائنی

    مثال:

    اگر x وڈا 0 تے y وڈا 0 {
    لکھو("دونوں مثبت ہیں")
}

    احتیاط

    • صفر سے تقسیم کرنے پر Runtime Error آ سکتا ہے۔
    • آپریٹرز کی ترجیح Parser کے قواعد کے مطابق ہوتی ہے۔
    • ہر لفظی آپریٹر کی دستیابی Lexer سے تصدیق کریں۔
