    شروعات — 5 منٹ میں SUPL

    1. پہلا پروگرام

    hello.slk بنائیں:

    پندھ_شروع "میرا پہلا پروگرام"
    لکھو("السلام علیکم دنیا!")
پندھ_مک_گیا

    چلائیں:

    saraiki hello.slk

    2. متغیرات

    پندھ_شروع "متغیرات"
    شے نام = "علی"
    شے عمر = 25

    لکھو("نام:", نام)
    لکھو("عمر:", عمر)
پندھ_مک_گیا

    3. حساب

    پندھ_شروع "حساب"
    شے a = 10
    شے b = 5

    لکھو(a جمع b)
    لکھو(a منفی b)
    لکھو(a ضرب b)
    لکھو(a ونڈ b)
پندھ_مک_گیا

    4. شرط

    پندھ_شروع "شرط"
    شے عمر = 20

    اگر عمر وڈا 18 {
        لکھو("بالغ")
    } ورنہ {
        لکھو("نابالغ")
    }
پندھ_مک_گیا

    5. لوپ

    پندھ_شروع "لوپ"
    برائے i میں 0 تک 5 {
        لکھو(i)
    }
پندھ_مک_گیا

    6. فنکشن

    پندھ_شروع "فنکشن"
    کم جمع(الف، ب) {
        واپس_ڈیو الف جمع ب
    }

    لکھو(جمع(5، 3))
پندھ_مک_گیا

    7. کلاس

    پندھ_شروع "کلاس"
    کلاس شخص {
        تعمیر(نام) {
            یہ.نام = نام
        }

        فنکشن دکھاؤ() {
            لکھو(یہ.نام)
        }
    }

    شے علی = نیا شخص("علی")
    علی.دکھاؤ()
پندھ_مک_گیا
